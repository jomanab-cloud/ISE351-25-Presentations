)

q1_result = {

Question 2: Attributes

Code 
viewof q2_answer = Inputs.checkbox(
  ["Primary key attributes", "Foreign key attributes", "Composite attributes", "Multi-valued attributes", "Derived attributes"],
  {
    label: "Which of the following are typically shown in E-R diagrams? (Select all that apply):",
    value: []
  }
)

q2_result = {
  if (q2_answer.length === 0) return "";
  
  const correct = ["Primary key attributes", "Composite attributes", "Multi-valued attributes"];
  const isCorrect = correct.every(item => q2_answer.includes(item)) && 
                   !q2_answer.includes("Foreign key attributes") && 
                   !q2_answer.includes("Derived attributes");
  
  if (isCorrect) {
    return html`<div style="color: green; font-weight: bold; margin-top: 10px;">
    ✓ Excellent! E-R diagrams show primary keys (underlined), composite attributes, and multi-valued attributes (double lines). Foreign keys are implied by relationships and derived attributes are usually not shown at the conceptual level.
    </div>`;
  } else {
    return html`<div style="color: orange; font-weight: bold; margin-top: 10px;">
    Partially correct. Remember: Foreign keys are not explicitly shown in E-R diagrams - they're implied by the relationships. Derived attributes are typically not included at the conceptual level.
    </div>`;
  }
}

Question 3: Cardinality Challenge

Code 
viewof q3_answer = Inputs.radio(
  ["1:1 (One-to-One)", "1:M (One-to-Many)", "M:N (Many-to-Many)", "Cannot be determined"],
  {
    label: "A STUDENT can enroll in many COURSES, and each COURSE can have many STUDENTS. What is the cardinality?",
    value: null
  }
)

q3_result = {
  if (q3_answer === null) return "";
  if (q3_answer === "M:N (Many-to-Many)") {
    return html`<div style="color: green; font-weight: bold; margin-top: 10px;">
    ✓ Perfect! This is a classic many-to-many relationship. In implementation, this would need a composite entity (like ENROLLMENT) to resolve it into two 1:M relationships.
    </div>`;
  } else {
    return html`<div style="color: red; font-weight: bold; margin-top: 10px;">

    </div>`;

}

Scenario Analysis Challenge

Real-World Scenario: A university library system

Code 
viewof scenario_answers = Inputs.form({
  entity_type: Inputs.radio(
    ["BOOK is strong, COPY is weak", "BOOK is weak, COPY is strong", "Both are strong", "Both are weak"],
    {label: "BOOK entity contains book information. COPY entity represents individual physical copies. Which is correct?"}
  ),
  
  cardinality: Inputs.radio(
    ["1:1", "1:M", "M:N", "Cannot determine"],
    {label: "One BOOK can have many COPIES, each COPY belongs to one BOOK. What's the cardinality?"}
  ),
  
  participation: Inputs.radio(
    ["BOOK optional, COPY mandatory", "BOOK mandatory, COPY optional", "Both mandatory", "Both optional"],
    {label: "Every COPY must be of some BOOK, but a BOOK might not have any COPIES yet (just ordered). Participation?"}
  )
})

scenario_result = {
  if (!scenario_answers.entity_type || !scenario_answers.cardinality || !scenario_answers.participation) {
    return "";
  }
  
  let score = 0;
  let feedback = [];
  
  if (scenario_answers.entity_type === "BOOK is strong, COPY is weak") {
    score++;
    feedback.push("✓ Correct! BOOK is strong (exists independently), COPY is weak (depends on BOOK)");
  } else {
    feedback.push("✗ Think about dependency: Can a copy exist without knowing which book it is?");
  }
  
  if (scenario_answers.cardinality === "1:M") {
    score++;
    feedback.push("✓ Right! One book to many copies");
  } else {
    feedback.push("✗ One book can have multiple copies, each copy is of one book");
  }
  
  if (scenario_answers.participation === "BOOK optional, COPY mandatory") {
    score++;
    feedback.push("✓ Perfect! COPY must relate to a BOOK (mandatory), but BOOK might not have copies yet (optional)");
  } else {
    feedback.push("✗ Consider: Must every copy belong to a book? Can a book exist without copies?");
  }
  
  const color = score === 3 ? "green" : score >= 2 ? "orange" : "red";
  const message = score === 3 ? "Excellent work!" : score >= 2 ? "Good job, review the missed concepts" : "Review the concepts and try again";
  
  return html`<div style="color: ${color}; font-weight: bold; margin-top: 15px;">
    <div>Score: ${score}/3 - ${message}</div>
    ${feedback.map(f => html`<div style="margin: 5px 0;">${f}</div>`)}
  </div>`;
}

Quick Fire Round!

30 seconds each - Think fast!

Code 
viewof quickfire = Inputs.form({
  q1: Inputs.radio(["True", "False"], {label: "Foreign keys are always shown in E-R diagrams"}),
  q2: Inputs.radio(["True", "False"], {label: "A recursive relationship is when an entity relates to itself"}),
  q3: Inputs.radio(["True", "False"], {label: "Cardinality (1,5) means minimum 1, maximum 5"}),
  q4: Inputs.radio(["True", "False"], {label: "Double-line rectangles represent strong entities"}),
  q5: Inputs.radio(["True", "False"], {label: "Multi-valued attributes are shown with double lines"})
})

quickfire_result = {
  if (!quickfire.q1 || !quickfire.q2 || !quickfire.q3 || !quickfire.q4 || !quickfire.q5) {
    return "";
  }
  
  const answers = ["False", "True", "True", "False", "True"];
  const responses = [quickfire.q1, quickfire.q2, quickfire.q3, quickfire.q4, quickfire.q5];
  const questions = [
    "Foreign keys are IMPLIED by relationships, not explicitly shown",
    "Correct! Like PART relating to PART in bill of materials",
    "Correct! Cardinality limits show min and max",
    "Double lines show WEAK entities, single lines show strong",
    "Correct! Double lines indicate multiple values possible"
  ];
  
  let score = 0;
  let feedback = [];
  
  responses.forEach((response, i) => {
    if (response === answers[i]) {
      score++;
      feedback.push(`✓ Q${i+1}: ${questions[i]}`);
    } else {
      feedback.push(`✗ Q${i+1}: ${questions[i]}`);
    }
  });
  
  const percentage = Math.round((score/5) * 100);
  const color = score >= 4 ? "green" : score >= 3 ? "orange" : "red";
  
  return html`<div style="color: ${color}; font-weight: bold; margin-top: 15px;">
    <div style="font-size: 1.2em;">Quick Fire Score: ${score}/5 (${percentage}%)</div>
    ${feedback.map(f => html`<div style="margin: 5px 0; font-size: 0.9em;">${f}</div>`)}
  </div>`;
}

Final Challenge: Design Time!

Scenario: A gym membership system

  • Members can join multiple gyms
  • Each gym has many members
  • Members attend classes
  • Each class has one instructor
  • Instructors can teach multiple classes
Code 
viewof design_challenge = Inputs.form({
  member_gym: Inputs.radio(["1:1", "1:M", "M:N"], {label: "MEMBER to GYM relationship:"}),
  member_class: Inputs.radio(["1:1", "1:M", "M:N"], {label: "MEMBER to CLASS relationship:"}),
  instructor_class: Inputs.radio(["1:1", "1:M", "M:N"], {label: "INSTRUCTOR to CLASS relationship:"}),
  entities_needed: Inputs.number({label: "How many entities would you need minimum?", min: 3, max: 10, step: 1})
})

design_result = {
  if (!design_challenge.member_gym || !design_challenge.member_class || !design_challenge.instructor_class || !design_challenge.entities_needed) {
    return "";
  }
  
  let score = 0;
  let feedback = [];
  
  if (design_challenge.member_gym === "M:N") {
    score++;
    feedback.push("✓ MEMBER-GYM: Correctly identified M:N relationship");
  } else {
    feedback.push("✗ MEMBER-GYM: Members can join multiple gyms, gyms have multiple members = M:N");
  }
  
  if (design_challenge.member_class === "M:N") {
    score++;
    feedback.push("✓ MEMBER-CLASS: Correctly identified M:N relationship");
  } else {
    feedback.push("✗ MEMBER-CLASS: Members attend multiple classes, classes have multiple members = M:N");
  }
  
  if (design_challenge.instructor_class === "1:M") {
    score++;
    feedback.push("✓ INSTRUCTOR-CLASS: Correctly identified 1:M relationship");
  } else {
    feedback.push("✗ INSTRUCTOR-CLASS: One instructor per class, but instructors can teach multiple classes = 1:M");
  }
  
  if (design_challenge.entities_needed >= 4 && design_challenge.entities_needed <= 6) {
    score++;
    feedback.push("✓ Good estimate! MEMBER, GYM, CLASS, INSTRUCTOR minimum (4), plus composite entities for M:N relationships");
  } else {
    feedback.push("✗ Think about: MEMBER, GYM, CLASS, INSTRUCTOR + composite entities for M:N relationships");
  }
  
  const color = score === 4 ? "green" : score >= 3 ? "orange" : "red";
  const message = score === 4 ? "Excellent design thinking!" : score >= 3 ? "Good analysis!" : "Review M:N relationships";
  
  return html`<div style="color: ${color}; font-weight: bold; margin-top: 15px;">
    <div>Design Score: ${score}/4 - ${message}</div>
    ${feedback.map(f => html`<div style="margin: 5px 0;">${f}</div>`)}
  </div>`;
}
